#version 330 core

in vec3 Normal;
in vec4 glPosition;
in vec3 glColor;
in vec2 TexC;

out vec4 FragColor;

uniform bool interpolation;
uniform sampler2D diffusemap;
uniform float opacity;
uniform vec3 viewPos;
uniform int numlight;
uniform vec3 tint;
uniform vec3 shaderColor;

struct Material {
    vec3 ambient;
    vec4 diffuse;
    vec3 specular;
    float shininess;
    bool hasdiffuse;
    bool hasspecular;
};

uniform Material material;

void main()
{
    vec4 color;

    if (material.hasdiffuse) {
        color = texture(diffusemap, TexC);
    } else if (interpolation) {
        color = vec4(glColor, 1.0);
    } else {
        color = vec4(shaderColor, 1.0);
    }

    FragColor = color * vec4(tint, 1.0);
    FragColor.a *= opacity;
}
